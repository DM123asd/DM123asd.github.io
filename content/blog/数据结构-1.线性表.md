### 线性表
- 由n个数据特性（数据类型）相同的元素构成的有限序列。
- 有唯一的首节点，除首节点外，每个元素都有一个前驱。
- 有唯一的尾节点，除尾节点外，每个元素都有一个后继。
### 顺序表
- 定义：一组连续的内存单元，依次储存线性表的各元素，即逻辑上相邻的元素，物理存储空间（储存位置）也相邻。比如数组。
- 顺序表初始化
```c
#include <stdio.h>
#include <stdlib.h>
#define MAXSIZE 100
typedef int ElemType;//定义int类型，方便后续更改数据类型

typedef struct
{
	ElemType data[MAXSIZE];
	int length;
}SeqList;//定义顺序表结构体，数组和长度

#方案1.初始化顺序表函数
void initList(SeqList *L)
{
	L->length = 0;
}
```
- 在尾部添加元素
```c
//尾部添加元素
int appendElem(SeqList *L, ElemType e)
{
	//如果数据表满了
	if (L->length>=MAXSIZE)
	{
		printf("顺序表已满\n");
		return 1;
	}
	//数据表没满，添加元素，长度+1
	L->data[L->length] = e;
	L->length ++;
	return 0;
}
```
- 遍历
```c
//遍历
void listElem(SeqList *L)
{
	for(int i = 0; i < L->length ; i++)
	{
		printf("%d",L->data[i]);//循环输出
	}
	printf("\n");
}
```
- 插入元素：元素从尾部起逐个向后移一位直到插入点，插入元素。
```c
//插入元素
int insertElem(SeqList *L, int pos , ElemType e)
{
	//如果顺序表满了
	if (L->length>=MAXSIZE)
	{
		printf("顺序表已满\n");
		return 1;
	}
	//如果插入位置大于顺序表
	if (pos > L->length + 1)
	{
		printf("插入位置无效\n");
		return 2;
	}
	//将插入点后的数，从尾部开始逐个后移一位
	for (int i = L->length - 1; i >= pos - 1;i-- )
	{
		//有前面的判断，i+1最大为MAXSIZE-1,不会溢出
		L->data[i+1]=L->data[i];
	}
	L->data[pos-1] = e;
	L->length ++;
	return 0;
}
```
- 删除：删除点后一位元素起逐个向前移一位，长度-1。
```c
int deleteElem(SeqList *L, int pos, ElemType *e)
{
	*e = L->data[pos-1];
	//如果删除位置大于顺序表
	if (pos > L->length)
	{
		printf("删除位置无效\n");
		return 1;
	}
	//pos位置起逐个向前移一位
	for (int i=pos; i<L->length; i++)
	{
		L->data[i-1]=L->data[i];
	}
	L->length--;
	return 0;
}
```
- 查找
```c
int findElem(SeqList *L, ElemType e)
{
	//判断顺序表不为空
	if(L->length == 0)
	{
		printf("空列表\n");
		return 0; 
	}
	//遍历查找
	for (int i=0; i<=L->length-1; i++)
	{
		if (L->data[i] == e)
		{
			return i+1;
		}
	}
	return 0;
}

```

- 动态分配内存地址初始化

```c
// 方案2. 动态分配内存地址初始化
typedef struct
{
    ElemType *data; // 此处改为指针，可动态扩容
    int length;
} SeqList; // 定义顺序表结构体

SeqList* initList()
{
    // 给结构体指针分配空间
    SeqList *L = (SeqList*)malloc(sizeof(SeqList));

    L->length = 0;
    return L;
}
```

### 线性链表-单向链表
- 用任意的存储单元储存线性表的数据元素。
- 每个节点包括两个域：
	1. 数据域：存储数据元素信息。
	2. 指针域：存储后继存储位置。
- 链表结构
```c
typedef int ElemType;
//节点结构：数据和指向下一节点的指针
typedef struct node
{
	ElemType data;
	struct node *next;
}Node;
```
- 单链表-初始化 
	- 头节点的指针指向null
```c
//初始化链表
Node *initList()
{
	//声明头节点，头节点数据为0，空链表头节点指向Null。
	Node *head = (Node*)malloc(sizeof(Node));
	head -> data = 0;
	head -> next = NULL;
	return head;
}

```
- 头插法
```c
//头插法
int insertHead(Node* L, ElemType e)
{
	//声明一个空节点，赋值，
	Node *p = (Node*)malloc(sizeof(Node));
	p->data = e;
	//节点的指针与头节点的指针指向同一节点
	p->next = L->next;
	//头节点的指针指向新节点
	L->next = p;
	return 0;
}
```
- 遍历
```c
//遍历
void listNode(Node *L)
{
	Node *p=L->next;
	while(p != NULL)
	{
		printf("%d\n",p->data);
		p=p->next;
	}
	printf("\n");
}
```

- 尾插法
```c
//1.遍历获取尾节点地址
Node* get_tail(Node *L)
{
	Node *p = L->next;
	while(p->next !=NULL)//此处注意为p->next!=NULL
	{
		p = p->next;
	}
	return p;//返回尾节点
}
//在尾节点插入新节点
Node* insertTail(Node *tail,ElemType e)
{
	Node *p = (Node*)malloc(sizeof(Node));
	p->data = e;
	tail->next = p;
	p->next = NULL;
	return p;//返回尾节点，后续可继续插入新节点
}
```
- 在指定位置插入数据
```c
//在指定位置插入数据
int insertNode(Node *L,int pos,ElemType e)
{
	Node *p = L;
	int i = 0;
	//找到插入位置的前驱节点
	while(i<pos -1)
	{
		p = p->next;
		i++;
		if (p== NULL)
		{
			return 1;
		}
	}
	//插入节点
	Node *q = (Node*)malloc(sizeof(Node));
	q->data = e;
	q->next = p->next;
	p->next = q;
	return 0;
}
```

- 删除节点
```c
//删除节点
int deleteNode(Node *L, int pos)
{
	Node *p = L;
	int i = 0;
	//找到待删除节点的前驱节点p
	while(i<pos -1)
	{
		p = p->next;
		i++;
		if (p== NULL)
		{
			return 1;
		}
	}
	//判断p有后继节点
	if(p->next== NULL)
	{
		printf("要删除的位置错误\n");
		return 2;
	}
	//删除并释放p的后继节点
	Node *q =  p->next;
	p->next=q->next;
	free(q);
	return 0;
}

```
- 获取链表长度
```c
//获取链表长度
int listLength(Node *L)
{
	Node *p = L->next;
	int len = 0;
	//链表长度不包括头节点
	while(p!=NULL)
	{
		len++;
		p = p->next;
	}
	return len;
}
```

- 释放链表
	1. 指针p指向头节点后的第一个节点
	2. 判断p是否为空。如果p不为空，指针q记录指针p的后继节点。
	3. 释放p指向的节点
	4. 指针p与指针q指向同一节点，重复循环。
```c
//释放链表
void freeList(Node *L)
{
	Node *p = L->next;
	Node *q;
	while(p != NULL)
	{
		q = p->next;
		free(p);
		p = q;
	}
	L->next = NULL;
}
```
- 调用
```c

int main()
{

	Node *list =  initList();
	//头插法
	insertHead(list,10);
	insertHead(list,20);

	Node *tail = get_tail(list);
	tail=insertTail(tail,30);
	tail=insertTail(tail,40);
	listNode(list);
	//中间插入节点
	insertNode(list,2,15);
	listNode(list);
	//删除节点
	deleteNode(list,2);
	listNode(list);
	printf("%d\n",listLength(list));
	//释放链表
	freeList(list);
	listNode(list);
	printf("%d\n",listLength(list));
    return 0;
}
```

### 线性链表-双向链表
- 除了头节点只有后指针，前指针为NULL，尾节点只有前指针，后指针为NULL。其余节点都有前后两个指针。
- 链表结构
```c
#include <stdio.h>
#include <stdlib.h>
typedef int ElemType;

typedef struct node{
	ElemType data;
	struct node *prev,*next;
}Node;//相比于单向链表，多了prev指针

//初始化链表
Node *initList()
{
	//声明头节点，头节点数据为0，空链表头节点指向Null。
	Node *head = (Node*)malloc(sizeof(Node));
	head -> data = 0;
	head -> next = NULL;
	head -> prev = NULL;
	return head;
}
```
- 头插法
	1. 新节点前指针指向头节点，后指针指向头节点的后继
	2. 判断原链表是否为空链表
		- 非空：头节点后一节点前指针指向新节点，头节点后指针指向新节点
		-  空：头节点后指针指向新节点
```c
//头插法
int insertHead(Node* L, ElemType e)
{
	Node *p = (Node*)malloc(sizeof(Node));
	p->data = e;
	p->next = L->next;
	p->prev = L;
	//判断原链表是否为空
	if(L->next != NULL)
	{
		L->next->prev = p;
	}
	L->next = p;
	return 0;
}
```
- 遍历：同单链表
```c
//遍历
void listNode(Node *L)

		printf("%d\n",p->data);

```
- 尾插法-同单向链表
```c
//1.遍历获取尾节点地址
Node* get_tail(Node *L)
{
	Node *p = L->next;
	while(p->next !=NULL)//p->next不为空即非尾结点
	{
		p = p->next;
	}

}
//2.在尾节点插入新节点
Node* insertTail(Node *tail,ElemType e)
{
	Node *p = (Node*)malloc(sizeof(Node));
	p->data = e;
	tail->next = p;
	p->next = NULL;

}
```
- 在指定位置插入
	1. 找到插入位置的前一节点，判断插入位置是否为有效值
	2. 插入元素，步骤如头插法
```c
int insertNode(Node *L,int pos, ElemType e)
{
	Node *p = L;
	int i = 0;
	//找得插入位置的前置节点
	while(i<pos-1)
	{
		p = p->next;
		i++;
		//无效位置不插入，return 1
		if(p == NULL)
		{
			return 1;
		}
	}
	//插入元素
	Node *q = (Node*)malloc(sizeof(Node));
	q->data = e;
	q->next = p->next;
	q->prev = p;
	if(p->next != NULL)
	{
		p->next->prev = q;
	}
	p->next = q;
	return 0;
}
```
- 删除节点
	1. 找到删除节点的前置节点p
	2. 用指针q记录要删除的节点
	3. p的后继节点改为q的后继节点
	4. 释放删除节点q的空间
```c
//删除
int deleteNode(Node *L, int pos)
{
	//找到删除位置的前置节点p
	Node *p = L;
	int i = 0;
	while(i<pos-1)
	{
		p = p->next;
		i++;
		if(p == NULL)
			return 1;
	}
	//判断q的位置是否有值
	if(p->next == NULL)
	{
		printf("要删除的位置错误\n");
		return 2;
	}
	//删除节点q
	Node *q = p->next;
	p->next = q->next;
	q->next->prev = p;
	free(q);
	return 0;
	
}
```
## 顺序表 VS 链表

| 比较维度      | 顺序表特点                             | 链表特点                        |
| --------- | --------------------------------- | --------------------------- |
| **存储空间**  | 预先分配，易出现闲置/溢出现象                   | 动态分配，无闲置/溢出现象               |
| **存储密度**  | 无需额外存储逻辑关系，存储密度=1                 | 需指针存逻辑关系，存储密度<1             |
| **存取元素**  | 随机存取，访问时间复杂度O(1)                  | 顺序存取，访问时间复杂度O(n)            |
| **插入/删除** | 需移动元素，时间复杂度O(n)                   | 无需移动元素（确定位置后），时间复杂度O(1)     |
| **适用场景**  | 1. 表长稳定且可预先确定范围<br>2. 少增删、频繁按位置访问 | 1. 表长变化较大<br>2. 频繁执行插入/删除操作 |

- 此处随机存取指无需顺序遍历，可直接按位置访问存取。
