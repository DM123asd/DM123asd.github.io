### 指针
- 指针是存放地址的变量。
- * 为间接引用符号，在非声明时，返回指针指向的地址的值，这个操作是解引用指针
```c++
#include <stdio.h>
using namespace std;

int main(int argc, char const *argv[])
{
    int a = 5; 
    int *p = &a; //定义指针，指针是存放地址的变量
	printf("%p\n",p);//指针指向的地址，即a的地址 0x7ffd0577447c
	printf("%p\n",&a);//a的地址 0x7ffd0577447c
	printf("%p\n",&p);//指针自己的地址 0x7ffd05774470
    printf("%d\n", *p); //指针指向地址的值
	//* 为间接引用符号，在非声明时，返回指针指向的地址的值，这个操作是解引用指针
    *p = 100;//可通过*p修改p指向的地址的值
    printf("%d\n", a);//a=100，a的值被修改
    return 0;
}
```

### 指针与函数
- 传值和传地址：
1. 引用变量为传值，不改变原变量数值
```c++
#include <stdio.h>
using namespace std;
void swap(int a,int b)
{
	int temp;
	temp = a;
	a = b;
	b = temp;
	printf("a=%d, b = %d\n", a, b);//a=10, b = 5
}
int main(int argc, char const *argv[])
{
	int m = 5;
	int n = 10;
	swap(m,n);
	printf("m=%d, n=%d\n",m,n);//m=5, n=10
}
```
2. 使用指针可改变地址的值，改变原变量的数值。
```c++
#include <stdio.h>
using namespace std;
void swap(int *a,int *b)//a,b为指针
{
	int temp;
	temp = *a;//a指向地址的值赋给temp
	*a = *b;//b指向地址的值赋给a的地址
	*b = temp;//temp赋给b指向的地址
	printf("a=%d, b = %d\n", *a, *b);//a=10, b = 5
}
int main(int argc, char const *argv[])
{
	int m = 5;
	int n = 10;
	swap(&m,&n);//传入地址
	printf("m=%d, n=%d\n",m,n);//m=10, n=5，变量被改变
}
```
### 指针与数组
```c++
#include <stdio.h>
using namespace std;
int main(int argc, char const *argv[])
{
	int a[] = {1,2,3,4,5};
	int *p;
	p = a;
	for(int i = 0; i<sizeof(a)/sizeof(a[0]);i++)
	{
		printf("%d\n",a[i]);//使用数组下标遍历
	}
	for(int i = 0; i<sizeof(a)/sizeof(a[0]);i++)
	{
		printf("%d\n",*(p+i));//使用指针遍历
	}
}
```
-  指针的算术运算：p+n→p+n*（对应数据类型字节数）
### 结构体与指针
- 指针高效访问和修改结构体：(\*pp).x, pp->x
```c++
#include <stdio.h>
using namespace std;

struct point
{
	int x;
	int y;
};

int main(int argc, char const *argv[])
{
	struct point p;
	p.x = 5;
	p.y = 10;
	struct point *pp;
	pp = &p;
	(*pp).x = 10;//利用指针给结构体赋值，可选(*pp).x,pp->x
	(*pp).y = 5;
	printf("x=%d,y=%d\n",p.x,p.y);//x=10,y=5
	printf("x=%d,y=%d\n",pp->x,pp->y);//x=10,y=5
}
```
