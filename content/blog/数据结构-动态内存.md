- 步骤：malloc函数分配内存->使用分配的内存->释放内存
- 案例1 - 基本数据类型
```c++
#include <stdio.h>
#include <stdlib.h>
using namespace std;

int main(int argc, char const *argv[])
{
	int *p;//声明指针
	p = (int*)malloc(sizeof(int));//给指针分配内存
	*p = 15;//使用分配的内存
	printf("%d\n",p);//35851888
	printf("%d\n",*p);//15
	
	free(p);//释放的是指针所指向的那块内存区域的使用权
	printf("%d\n",p);//p值不变
	printf("%d\n",*p);//0 
    return 0;
}
```
- 案例2-字符串类型
```c++
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
using namespace std;

int main(int argc, char const *argv[])
{
	char *s;
	s = (char*)malloc(10);//申请了10个字节空间，可放9个字符+/0
	strcpy(s,"Hello");//strcpy将一个字符串复制到另一块内存地址开始的空间,需要目标内存的起始地址
	printf("%s\n",s);//用 %s 输出整个字符串，传递地址 s
	free(s);
    return 0;
}
```
- 案例3-数组类型
```c++
#include <stdio.h>
#include <stdlib.h>
using namespace std;

int main(int argc, char const *argv[])
{
	int *arr = (int*)malloc(5 * sizeof(int));
	for (int i=0;i<5;i++)
	{
		arr[i] = i;
		printf("%d\n",arr[i]);
		
	}
	free(arr);
    return 0;
}
```
- 案例4- 结构体
```c++
#include <stdio.h>
#include <stdlib.h>
using namespace std;

typedef struct 
{
	int x;
	int y;
}po;

int main(int argc, char const *argv[])
{
	po *p;
	p = (po*)malloc(sizeof(po));//给指针分配内存
	p->x = 5;
	p->y = 10;
	printf("%d\n",p->x);
	printf("%d\n",p->y); 
    return 0;
}
```
