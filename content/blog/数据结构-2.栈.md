### 栈
- 只能在表尾进行插入或删除操作的线性表。
- 表尾是栈顶，表头是栈底。先进后出，后进先出。
- 基本操作：进栈(push)，出栈(pop)。
#### 栈的顺序结构
- 初始化
```c
#include <stdio.h>
#include <stdlib.h>

#define MAXSIZE 100
typedef int ElemType;
typedef struct{
	ElemType data[MAXSIZE];
	int top;//指示栈顶的位置
}Stack;

void initStack(Stack *s)
{
	s->top = -1;//栈顶为-1时为空
}
```
- 判断栈是否为空
```c
int isEmpty(Stack *s)
{
	if(s->top == -1)
	{
		printf("空栈\n");
		return 1;
	}
	else
	{
		return 0;
	}//空栈返回1，非空返回0
}
```
- 进栈/压栈
```c
int push(Stack *s,ElemType e)
{
	//判断栈有没有满
	if(s->top >= MAXSIZE-1)
	{
		printf("满了\n");
		return 1;
	}
	//进栈
	s->top++;
	s->data[s->top] = e;
	return 0;
}
```
- 出栈
```c
int pop(Stack *s, ElemType *e)
{
	//判断栈是否为空
	if(s->top == -1)
	{
		printf("空的\n");
		return 1;
	}
	*e=s->data[s->top];
	s->top--;
	return 0;
}
```
- 获取栈顶元素
```c
int getTop(Stack *s,ElemType *e)
{
	if(s->top == -1)
	{
		printf("空的\n");
		return 1;
	}
	*e = s->data[s->top];
	return 0;
}
```
- 调用
```c
int main(int argc, char const *argv[])
{
	//创建栈s
	Stack s;
	initStack(&s);
	//压栈
	push(&s,10);
	push(&s,20);
	push(&s,30);
	//出栈
	ElemType e;
	pop(&s,&e);
	printf("%d\n",e);
	//获取栈顶元素
	getTop(&s,&e);
	printf("%d\n",e);
	return 0;	
}
```

#### 栈的链式结构
- 把头结点作为栈顶，尾结点看做栈底，只对头结点操作。
- 栈的链式结构-初始化
```c
typedef int ElemType;
//每个节点有一个数据和指向下一个节点的指针
typedef struct stack
{
	ElemType data;
	struct stack *next;
}Stack;
Stack *initStack()
{
	Stack *s = (Stack*)malloc(sizeof(Stack));
	s->data = 0;
	s->next = NULL;
	return s;
}
```
判断栈是否为空
```c
int isEmpty(Stack *s)
{
	if(s->next == NULL)
	{
		printf("空的\n");
		return 1;
	}
	else
	{
		return 0;
	}
}
```
-  进栈/压栈 
```c
//头插法
int push(Stack *s,ElemType e)
{
	Stack *p = (Stack*)malloc(sizeof(Stack));
	p->data = e;
	p->next= s->next;
	s->next = p;
	return 0;
}
```
- 出栈：删除头结点后的一个结点
```c
int pop(Stack *s,ElemType *e)
{
	if(s->next == NULL)
	{
		printf("空的\n");
		return 1;
	}
	*e = s->next->data;
	Stack *q = s->next;
	s->next = q->next;
	free(q);
	return 0;
}
```
- 获取栈顶元素：头结点后的一个结点的data
```c
int getTop(Stack *s,ElemType *e)
{
	if(s->next == NULL)
	{
		printf("空的\n");
		return 1;
	}
	*e = s->next->data;
	return 0;
}
```
- 调用
```c
int main(int argc,char const *argv[])
{
	Stack *s = initStack();
	//进栈
	push(s,10);
	push(s,20);
	push(s,30);
	//出栈
	ElemType e;
	pop(s,&e);
	printf("%d\n",e);
	//获取栈顶元素
	getTop(s,&e);
	printf("%d\n",e);
	return 0;
}
```
#### 栈的应用- 表达式求值
```c
#include <stdio.h>
#include <stdlib.h>

#define MAXSIZE 100
typedef int ElemType;
typedef struct
{
	ElemType *data;
	int top;
}Stack;

typedef enum
{
	LEFT_PARE,RIGHT_PARE,ADD,SUB,
	MUL,DIV,MOD,EOS,NUM
}contentType;

Stack* initStack()
{
	Stack *s = (Stack*)malloc(sizeof(Stack));
	s->data = (ElemType*)malloc(sizeof(ElemType)*MAXSIZE);
	s->top = -1;
	return s;
}

int push(Stack *s,ElemType e)
{
	//判断栈有没有满
	if(s->top >= MAXSIZE-1)
	{
		printf("满了\n");
		return 1;
	}
	//进栈
	s->top++;
	s->data[s->top] = e;
	return 0;
}

int pop(Stack *s, ElemType *e)
{
	//判断栈是否为空
	if(s->top == -1)
	{
		printf("空的\n");
		return 1;
	}
	*e=s->data[s->top];
	s->top--;
	return 0;
}

int getTop(Stack *s,ElemType *e)
{
	if(s->top == -1)
	{
		printf("空的\n");
		return 1;
	}
	*e = s->data[s->top];
	return 0;
}

char expr[] = "82/2+56*-";
//识别字符类型
contentType getToken(char *symbol,int *index)
{
	*symbol = expr[*index];
	*index = *index+1;
	//判断symbol指针所指部分为什么符号
	switch(*symbol)
	{
		case '(':
			return LEFT_PARE;
		case ')':
			return RIGHT_PARE;
		case '+':
			return ADD;
		case '-':
			return SUB;
		case '*':
			return MUL;
		case '/':
			return DIV;
		case '%':
			return MOD;
		case '\0':
			return EOS;
		default:
			return NUM;
	}
}
int eval(Stack *s)
{
	char symbol;
	int op1,op2;
	int index = 0;
	contentType token;
	token = getToken(&symbol,&index);
	ElemType result;
	
	while(token!=EOS)//非空
	{
		//如果是数字，数字压栈
		if(token == NUM)
		{
			//字符数字转换为对应的整数值，然后将其压入栈中
			//如果 symbol是 '5'，其ASCII码为53。'5' - '0'就相当于 53 - 48
			push(s,symbol - '0');
		}
		//如果不是数字，数字出栈，操作符计算后压栈，其余结束进程
		else
		{
			pop(s,&op2);
			pop(s,&op1);
			
			switch(token)
			{
				case ADD:
					push(s,op1+op2);
					break;
				case SUB:
					push(s,op1-op2);
					break;
				case MUL:
					push(s,op1*op2);
					break;
				case DIV:
					push(s,op1/op2);
					break;
				case MOD:
					push(s,op1%op2);
					break;
				default:
					break;
			}
		}
		//继续循环读取token
		token= getToken(&symbol,&index);
	}
	//输出结果
	pop(s,&result);
	printf("%d\n",result);
	return 0;
}

int main(int argc, char const *argv[])
{
	//创建栈s
	Stack *s = initStack();
	//调用
	eval(s);
	return 0;	
}
```
