### 结构体
```c++
struct 结构体名
{
	数据类型 变量名;
	数据类型 变量名;
	...
};

struct 结构体名 变量名//调用
变量名.结构体内部变量名 = 值//赋值
```
- 例1：结构体的使用
```c++
#include <stdio.h>
using namespace std;

struct point
{
	int x;
	int y;
};//定义结构体
int main(int argc, char const *argv[])
{
	struct point p;//结构体初始化和调用
	p.x=5;//结构体内部变量赋值
	p.y=10;
	printf("%d\n",p.x);//5
	printf("%d\n",p.y);//10
}
```
- 例2：编写函数，传入int类型x,y参数，返回point结构体
```c++
struct point
{
	int x;
	int y;
};
struct point createPoint(int x,int y)
{
	struct point temp;
	temp.x = x;
	temp.y = y;
	return temp;
}//可通过createPoint快速生成point
int main(int argc, char const *argv[])
{
	struct point p;//创建结构体p
	p=createPoint(5,10);//将函数返回的point赋值给p
	printf("%d\n",p.x);//5
	printf("%d\n",p.y);//10
}
```
### 结构体与指针
- 指针高效访问和修改结构体：(\*pp).x, pp->x
```c++
#include <stdio.h>
using namespace std;

struct point
{
	int x;
	int y;
};

int main(int argc, char const *argv[])
{
	struct point p;
	p.x = 5;
	p.y = 10;
	struct point *pp;
	pp = &p;
	(*pp).x = 10;//利用指针给结构体赋值，可选(*pp).x,pp->x
	(*pp).y = 5;
	printf("x=%d,y=%d\n",p.x,p.y);//x=10,y=5
	printf("x=%d,y=%d\n",pp->x,pp->y);//x=10,y=5
}
```
### 类型定义
```c++
typedef 数据类型 别名;
typedef int zx;
```
- 用 typedef 定义struct，代码更简洁，声明变量不需要重复用struct
```c++
typedef struct 结构体名
{
	数据类型 变量名;
	数据类型 变量名;
	...
}别名;

typedef struct 
{
	数据类型 变量名;
	数据类型 变量名;
	...
}别名;
```
